-- Determines if a given year is a leap year
isLeapYear :: Int -> Bool
isLeapYear year
    =  (year `mod` 400 == 0)
    || (year `mod` 4 == 0 && year `mod` 100 /= 0)

{-
    - Determines how many days are in a given month,
      respecting if the current year is a leap year
-}
daysInMonth :: Int -> Int -> Int
daysInMonth month year
    = case month of
      1  -> 31
      2  -> if (isLeapYear year) then 29 else 28
      3  -> 31
      4  -> 30
      5  -> 31
      6  -> 30
      7  -> 31
      8  -> 31
      9  -> 30
      10 -> 31
      11 -> 30
      12 -> 31

{-
    - Generates day,month,year 3-tuples starting after 2025.
    - Since teaser says forever, I have set a temp limit of 2040
-}
generator :: [(Int,Int,Int)]
generator
    = [(day,month,year)
    | year <- [2026..2040]
    , month <- [1..12]
    , let maxDays = daysInMonth month year
    , day <- [1..maxDays]
    ]

{-
    - Given month, if single digit, pad with leading 0,
      otherwise, simply return
-}
pad :: String -> String
pad n
    = if (read n < 10) then ("0" ++ n) else n

{-
	- Converts (day,month,year) 3-tuple to string
-}
display :: (Int,Int,Int) -> String
display (day,month,year)
    = (show day) ++ (pad (show month)) ++ (show year)

-- Determines if an Int is a perfect square
perfectSquare :: Int -> Bool
perfectSquare n
    = floor srn == ceiling srn
    where
    srn = sqrt (fromIntegral n)

selector :: (Int,Int,Int) -> Bool
selector (day,month,year)
    = perfectSquare (read (display (day,month,year)))

main :: IO ()
main = print(filter selector generator)
