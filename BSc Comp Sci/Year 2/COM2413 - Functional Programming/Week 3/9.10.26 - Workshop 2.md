
Sunday Times Teaser 3154 of 5th March 2023, Gill’s Primes by Bill Kinally.

```Haskell
{- 
    Given 3 ints: A,B,C
    99 < A < B < C < 1000
    B - A == C - B
    prime(A), prime(B), prime(C)
-}

import Data.List

prime :: Int -> Bool
prime n
    = factors n == [1,n]
    where
    factors n
        = [f | f <- [1..n], n `mod` f == 0]


generator :: [(Int,Int,Int)]
generator
    = [(read [a1,a2,a3], read [b1,b2,b3], read [c1,c2,c3])
    | a1 <- "123456789"
    , a2 <- "0123456789" \\ [a1]
    , a3 <- "1379" \\ [a1,a2]
    , b1 <- "123456789" \\ [a1,a2,a3]
    , b2 <- "0123456789" \\ [a1,a2,a3,b1]
    , b3 <- "1379" \\ [a1,a2,a3,b1,b2]
    , c1 <- "123456789" \\ [a1,a2,a3,b1,b2,b3]
    , c2 <- "0123456789" \\ [a1,a2,a3,b1,b2,b3,c1]
    , c3 <- "1379" \\ [a1,a2,a3,b1,b2,b3,c1,c2]
    ]

selector :: (Int,Int,Int) -> Bool
selector (a,b,c)
    = a < b && b < c
    && b - a == c - b
    && prime a && prime b && prime c

main :: IO ()
main = print(filter selector generator)

-- OUTPUT ------------------------------------------------------------------------
[(109,283,457),(157,283,409)]
```

There are two answers, but Teasers always have a single answer, ONLY!

So, the hint is in the riddle at the end: "The sum of the digits for each 3-digit number is odd."

This means the answer is `(157,283,409)` because the sum of `109` is `10` which is even.
