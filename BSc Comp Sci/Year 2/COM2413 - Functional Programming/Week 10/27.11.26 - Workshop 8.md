data Base
  = Adenine
  | Cytosine
  | Guanine
  | Thymine

instance Show Base where
  showsPrec p a
    = case a of
        Adenine  -> showChar 'A'
        Cytosine -> showChar 'C'
        Guanine  -> showChar 'G'
        Thymine  -> showChar 'T'

  showList []
    = id
  showList (x:xs)
    = shows x . showList xs

instance Read Base where
  readsPrec p (c:cs)
    = case c of
        'A' -> [(Adenine, cs)]
        'C' -> [(Cytosine,cs)]
        'G' -> [(Guanine, cs)]
        'T' -> [(Thymine, cs)]

  readList []
    = [([],[])]
  readList (c:cs)
    = [(x:xs,cs2) | (x,cs1) <- reads (c:cs), (xs,cs2) <- reads cs1]

data Op b
  = Insert b
  | Delete b
  | Change b b
    deriving Show

edits :: [b] -> [b] -> [[Op b]]
edits (a:as) (b:bs)
  = [Change a b : ops | ops <- edits as bs]
    ++
    [Delete a   : ops | ops <- edits as (b:bs)]
    ++
    [Insert b   : ops | ops <- edits (a:as) bs]
edits as []
  = [[Delete a | a <- as]]
edits [] bs
  = [[Insert b | b <- bs]]

column :: Show b => Op b -> (String,String)
column (Insert p)
    = ("-", show p)
column (Delete p)
    = (show p, "-")
column (Change p q)
    = (show p, show q)
