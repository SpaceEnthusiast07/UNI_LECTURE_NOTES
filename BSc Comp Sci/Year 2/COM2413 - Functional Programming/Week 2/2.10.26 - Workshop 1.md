import Data.List

-- Converts a base 10 number to base 5
toBase5 :: Int -> [Int]
toBase5 n
    | n < 5     = [n]
    | otherwise = toBase5 d ++ [m]
    where
    d = n `div` 5
    m = n `mod` 5

-- Converts a base 10 number to base 8
toBase8 :: Int -> [Int]
toBase8 n
    | n < 8     = [n]
    | otherwise = toBase8 d ++ [m]
    where
    d = n `div` 8
    m = n `mod` 8

generator :: [Int]
generator = [0..624]

selector :: Int -> Bool
selector n
    = sort t == sort u
    where
    t = toBase5 n
    u = toBase8 n

main :: IO ()
main = print(filter selector generator)

toBase16 :: Int -> [Char]
toBase16 n
    | n < 16    = toChar16 n
    | otherwise = (toBase16 d) ++ (toChar16 m)
    where
    d = n `div` 16
    m = n `mod` 16
